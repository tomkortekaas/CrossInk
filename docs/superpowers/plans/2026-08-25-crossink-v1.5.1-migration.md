```cpp
  // Drive gated peripheral rails to their inactive level and latch them so the
  // level survives deep sleep (gpio_hold_en per pin below; PowerManager::deepSleep()
  // enables gpio_deep_sleep_hold_en(), which makes the holds persist). Without
  // this, boards with gated rails (e.g. Sticky: GT911 on TP_PWR_EN, SD on
  // SD_PWR_EN, EPD on EP_PWR_EN) leave those peripherals powered all through
  // deep sleep — milliamps of standby drain. Call after the display driver's
  // deep-sleep command and before deepSleep(); wake is a chip reset, so rails
  // re-enable in the normal init path. NOTE: cutting the touch rail forfeits
  // touch-to-wake. A rail-enable pin that is also a power latch is skipped: on
  // the Xteink X3/X4 the "SD enable" (GPIO13) is the battery MOSFET gate, so
  // cutting it here would power the board off rather than gate a peripheral.
  // Releasing a latch is the caller's decision — see holdPowerRailsForTimerWake().
  static void powerDownRailsForSleep();

  // Drive the active board's power-latch pins to their asserted (HIGH) level and
  // hold them through deep sleep. A timer wake only happens if the board is still
  // powered when the timer fires, so any sleep that arms one must call this: on a
  // battery-latched board an unheld latch pin floats once the GPIOs are isolated,
  // the latch node discharges, and the device switches off instead of sleeping.
  // USB masks this entirely, because VBUS feeds the same latch node in parallel.
  static void holdPowerRailsForTimerWake();
```

- [ ] **Step 3: Edit `PowerManager.cpp` — merge the helper functions**

Find the `holdRailOff` helper in the anonymous namespace (upstream may have kept the name

```cpp
void holdRail(int8_t pin, uint8_t level) {
  if (pin < 0) return;
  const auto g = static_cast<gpio_num_t>(pin);
  gpio_hold_dis(g);
  pinMode(pin, OUTPUT);
  digitalWrite(pin, level);
  gpio_hold_en(g);
}

// True when a rail-enable pin doubles as a power latch. Cutting such a pin does
// not gate a peripheral, it powers the board off (Xteink X3/X4: GPIO13 is both).
bool pinIsPowerLatch(int8_t pin) {
  if (pin < 0) return false;
  const auto& p = BoardConfig::ACTIVE.power;
  return pin == p.latch0 || pin == p.latch1;
}

// Cut a peripheral rail, unless the pin is really the board's power latch.
void holdPeripheralRailOff(int8_t pin, uint8_t offLevel) {
  if (pinIsPowerLatch(pin)) return;
  holdRail(pin, offLevel);
}
```

void PowerManager::powerDownRailsForSleep() {
  const auto& b = BoardConfig::ACTIVE;
  // [keep upstream's existing comment block explaining resetSleepLevel here]
  const uint8_t resetSleepLevel = b.display.powerEnable >= 0 ? LOW : HIGH;
  holdRail(b.display.rst, resetSleepLevel);
  holdPeripheralRailOff(b.display.powerEnable, LOW);
  // SD enable OFF = the inactive level: LOW for active-high enables, HIGH for the

  holdPeripheralRailOff(b.sd.powerEnable, b.sd.powerActiveHigh ? LOW : HIGH);
  holdPeripheralRailOff(b.touch.powerEnable, b.touch.powerEnableActiveHigh ? LOW : HIGH);
  // The mic enable also carries a polarity flag; OFF is the inactive level.
  holdPeripheralRailOff(b.mic.enable, b.mic.enableActiveHigh ? LOW : HIGH);
}

void PowerManager::holdPowerRailsForTimerWake() {
